    2. Create the users settings component using the same pattern as sibling settings pages:

       If view-based Volt (⚡ pattern):
       ```blade
       <?php
       use App\Models\User;
       use App\Models\UserInvitation;
       use App\Notifications\UserInvitedNotification;
       use Illuminate\Support\Facades\Password;
       use Illuminate\Support\Str;
       use function Livewire\Volt\{state, action, computed};

       if (! auth()->user()->is_admin) {
           abort(403);
       }

       state(['email' => '']);

       $invite = action(function () {
           $this->validate(['email' => ['required', 'email', 'unique:users,email']]);

           $pending = UserInvitation::valid()->where('email', $this->email)->exists();
           if ($pending) {
               $this->addError('email', 'An invitation is already pending for this email.');
               return;
           }

           $invitation = UserInvitation::create([
               'email' => $this->email,
               'token' => Str::random(64),
               'expires_at' => now()->addHours(72),
           ]);

           \Illuminate\Notifications\AnonymousNotifiable::route('mail', $this->email)
               ->notify(new UserInvitedNotification($invitation));

           $this->reset('email');
           session()->flash('invite_sent', 'Invitation sent to ' . $this->email);
       });

       $deleteUser = action(function (int $userId) {
           $user = User::findOrFail($userId);

           if ($user->id === auth()->id()) {
               return; // Cannot delete yourself
           }

           $user->delete();
       });

       $resetPassword = action(function (int $userId) {
           $user = User::findOrFail($userId);

           Password::sendResetLink(['email' => $user->email]);

           session()->flash('reset_sent', 'Password reset email sent to ' . $user->email);
       });

       $users = computed(fn () => User::where('id', '!=', auth()->id())->orderBy('name')->get());
       ?>
       ```
