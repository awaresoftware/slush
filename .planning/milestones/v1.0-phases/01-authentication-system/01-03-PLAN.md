    2. Add routes to routes/web.php (OUTSIDE any auth middleware groups):
       ```php
       Route::get('/invite/{token}', function (string $token) {
           $invitation = \App\Models\UserInvitation::where('token', $token)->first();

           if (! $invitation || $invitation->isExpired() || $invitation->isUsed()) {
               abort(404);
           }

           return view('pages.auth.accept-invite', compact('invitation'));
       })->name('invite.accept');

       Route::post('/invite/{token}', function (\Illuminate\Http\Request $request, string $token) {
           $invitation = \App\Models\UserInvitation::where('token', $token)->first();

           if (! $invitation || $invitation->isExpired() || $invitation->isUsed()) {
               abort(404);
           }

           $request->validate([
               'password' => ['required', 'string', 'confirmed', \Illuminate\Validation\Rules\Password::defaults()],
           ]);

           $user = \App\Models\User::create([
               'name' => explode('@', $invitation->email)[0],  // derive name from email prefix
               'email' => $invitation->email,
               'password' => $request->password,
               'is_admin' => false,
           ]);

           $invitation->update(['accepted_at' => now()]);

           \Illuminate\Support\Facades\Auth::login($user);

           return redirect()->route('dashboard');
       })->name('invite.store');
       ```
       Note: name the routes invite.accept and invite.store consistently.
