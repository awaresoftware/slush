test('theme toggle saves dark mode preference to database', function () {
    $user = User::factory()->create(['theme' => 'spotify-dark']);
    $this->actingAs($user);

    Livewire::test('pages::settings.appearance')
        ->set('isDarkMode', false)
        ->assertHasNoErrors();

    $user->refresh();
    expect($user->theme)->toBe('light');
});

test('theme toggle saves light to dark preference to database', function () {
    $user = User::factory()->create(['theme' => 'light']);
    $this->actingAs($user);

    Livewire::test('pages::settings.appearance')
        ->set('isDarkMode', true)
        ->assertHasNoErrors();

    $user->refresh();
    expect($user->theme)->toBe('spotify-dark');
});

test('appearance page mounts with correct dark mode state', function () {
    $user = User::factory()->create(['theme' => 'light']);
    $this->actingAs($user);

    Livewire::test('pages::settings.appearance')
        ->assertSet('isDarkMode', false);
});
