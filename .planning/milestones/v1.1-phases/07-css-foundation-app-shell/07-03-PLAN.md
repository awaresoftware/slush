<interfaces>
<!-- Mary UI component APIs (from vendor source, verified in RESEARCH.md) -->

x-mary-main:
  - x-slot:sidebar — accepts drawer="main-drawer", collapsible, class
  - x-slot:content — main content area, class merges with defaults
  - x-slot:footer — optional footer

x-mary-menu:
  - activate-by-route — auto-marks items active based on URL
  - Children: x-mary-menu-item

x-mary-menu-item:
  - title="Text" — display text
  - icon="o-icon-name" — Heroicon with prefix
  - link="url" — navigation URL (auto-adds wire:navigate)
  - Do NOT manually add wire:navigate (it's automatic)

x-mary-dropdown:
  - x-slot:trigger — any element as trigger
  - Children: <li> elements for menu items
  - Positioning: default bottom-start

x-mary-nav:
  - sticky — makes nav sticky
  - x-slot:brand — logo area
  - x-slot:actions — right-side actions

x-mary-icon:
  - name="o-icon-name" — Heroicon outline prefix
  - class="size-N" — sizing

<!-- Routes available (from existing sidebar) -->
route('dashboard') — Dashboard page
route('library.index') — Library page
route('playlists.index') — Playlists page
route('profile.show', auth()->user()) — Profile page
route('profile.edit') — Settings page
route('logout') — Logout (POST)
</interfaces>
