```php
<?php

namespace App\Http\Controllers;

use Illuminate\Http\JsonResponse;
use Illuminate\Support\Facades\DB;

class HealthCheckController extends Controller
{
    /**
     * Database connectivity timeout in seconds (D-05).
     */
    private const DB_TIMEOUT_SECONDS = 3;

    public function __invoke(): JsonResponse
    {
        $dbStatus = 'ok';
        $dbLatency = null;

        $previousTimeout = ini_get('default_socket_timeout');

        try {
            ini_set('default_socket_timeout', self::DB_TIMEOUT_SECONDS);

            $start = microtime(true);
            DB::connection()->getPdo();
            $dbLatency = (int) round((microtime(true) - $start) * 1000);
        } catch (\Throwable $e) {
            $dbStatus = 'error';
        } finally {
            ini_set('default_socket_timeout', $previousTimeout);
        }

        $status = $dbStatus === 'ok' ? 'ok' : 'error';
        $httpCode = $status === 'ok' ? 200 : 503;

        return response()->json([
            'status' => $status,
            'version' => config('app.version'),
            'php' => PHP_VERSION,
            'database' => [
                'status' => $dbStatus,
                'latency_ms' => $dbLatency,
            ],
        ], $httpCode);
    }
}
```

Key points about the timeout implementation (D-05):
- Sets `ini_set('default_socket_timeout', 3)` BEFORE the `DB::connection()->getPdo()` call to enforce a 3-second timeout on the database connectivity check.
- Uses a `finally` block to restore the previous `default_socket_timeout` value, preventing side effects on subsequent requests in the same process.
- The timeout is defined as a class constant `DB_TIMEOUT_SECONDS` for clarity and maintainability.
- The `microtime(true)` start is measured AFTER the timeout is set, inside the try block, so latency measurement is accurate.
