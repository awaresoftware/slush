if [ ! -z "$WWWUSER" ]; then
    usermod -u $WWWUSER sail
fi
if [ $# -gt 0 ]; then
    if [ "$SUPERVISOR_PHP_USER" = "root" ]; then
        exec "$@"
    else
        exec gosu $WWWUSER "$@"
    fi
else
    exec /usr/bin/supervisord -c /etc/supervisor/conf.d/supervisord.conf
fi

**Create `compose.yaml`** at project root with these exact services:

```yaml
# For more information: https://laravel.com/docs/sail
services:
    laravel.test:
        build:
            context: ./vendor/laravel/sail/runtimes/8.4
            dockerfile: Dockerfile
            args:
                WWWGROUP: '${WWWGROUP}'
        image: sail-8.4/app
        extra_hosts:
            - 'host.docker.internal:host-gateway'
        ports:
            - '${APP_PORT:-80}:80'
        environment:
            WWWUSER: '${WWWUSER}'
            LARAVEL_SAIL: 1
            XDEBUG_MODE: '${SAIL_XDEBUG_MODE:-off}'
            XDEBUG_CONFIG: '${SAIL_XDEBUG_CONFIG:-client_host=host.docker.internal}'
            IGNITION_LOCAL_SITES_PATH: '${PWD}'
        volumes:
            - '.:/var/www/html'
        networks:
            - sail
        depends_on:
            mysql:
                condition: service_healthy
            redis:
                condition: service_healthy
        command: ["/bin/bash", "/var/www/html/docker/start.sh"]
    worker:
        image: sail-8.4/app
        command: ["php", "artisan", "queue:work", "--sleep=3", "--tries=3", "--max-time=3600"]
        volumes:
            - '.:/var/www/html'
        networks:
            - sail
        depends_on:
            laravel.test:
                condition: service_started
            mysql:
                condition: service_healthy
            redis:
                condition: service_healthy
        restart: unless-stopped
        environment:
            WWWUSER: '${WWWUSER}'
            LARAVEL_SAIL: 1
    mysql:
        image: 'mysql:8.4'
        ports:
            - '${FORWARD_DB_PORT:-3306}:3306'
        environment:
            MYSQL_ROOT_PASSWORD: '${DB_PASSWORD}'
            MYSQL_ROOT_HOST: '%'
            MYSQL_DATABASE: '${DB_DATABASE}'
            MYSQL_USER: '${DB_USERNAME}'
            MYSQL_PASSWORD: '${DB_PASSWORD}'
            MYSQL_ALLOW_EMPTY_PASSWORD: 1
        volumes:
            - 'sail-mysql:/var/lib/mysql'
            - './vendor/laravel/sail/database/mysql/create-testing-database.sh:/docker-entrypoint-initdb.d/10-create-testing-database.sh'
        networks:
            - sail
        healthcheck:
            test: ["CMD", "mysqladmin", "ping", "-p${DB_PASSWORD}"]
            retries: 5
            timeout: 5s
            interval: 10s
            start_period: 30s
    redis:
        image: 'redis:alpine'
        ports:
            - '${FORWARD_REDIS_PORT:-6379}:6379'
        volumes:
            - 'sail-redis:/data'
        networks:
            - sail
        healthcheck:
            test: ["CMD", "redis-cli", "ping"]
            retries: 3
            timeout: 5s
networks:
    sail:
        driver: bridge
volumes:
    sail-mysql:
        driver: local
    sail-redis:
        driver: local
```

Key decisions:
- NO `version:` header (deprecated in Docker Compose v2)
- NO VITE_PORT mapping (D-03: assets are pre-built, no Vite dev server in Docker)
- NO Mailpit or Meilisearch (D-06: minimal services)
- MySQL healthcheck has `start_period: 30s`, `retries: 5`, `interval: 10s` (Pitfall 1 prevention)
- Worker does NOT call docker/start.sh (Pitfall 3 prevention) — runs queue:work directly
- Worker depends on laravel.test with `service_started` (web container runs migrations first)
- Port configurable via `APP_PORT` env var (D-07) with default 80 (D-08)

**Create `docker/start.sh`** (must be executable):

```bash
#!/usr/bin/env bash
set -e

cd /var/www/html

# Generate APP_KEY if missing (same guard as composer run setup)
php -r "\$e=file_exists('.env')?file_get_contents('.env'):'';preg_match('/^APP_KEY=base64:/m',\$e)||passthru('php artisan key:generate --force --no-interaction');"

# Run migrations on every boot — idempotent, picks up new migrations after git pull (D-02)
php artisan migrate --force

# Build frontend assets only when manifest is missing — skip on subsequent restarts (D-03)
if [ ! -f /var/www/html/public/build/manifest.json ]; then
    echo "Building frontend assets (first run)..."
    npm install
    npm run build
fi

# Create storage symlink — idempotent with --force
php artisan storage:link --force

# Hand off to supervisord (starts PHP web server via Sail's config)
exec /usr/bin/supervisord -c /etc/supervisor/conf.d/supervisord.conf
```

After creating both files, run `chmod +x docker/start.sh`.
